# Workspace Lifecycle Engine — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the demo-coupled Import Engine with a manifest-driven, plugin-extensible Workspace Lifecycle Engine that is the single entry point for every workspace lifecycle operation (CREATE / IMPORT / SYNC / REFRESH / VALIDATE).

**Architecture:** The engine is structured as a plugin-based Dataset Registry, a set of ten independently callable Pipeline Stages, and a Lifecycle Engine that routes operations to the correct subset of stages. Every operation emits progress events over the existing WebSocket. Future dataset producers register themselves via `registerDatasetType()` — zero engine changes required.

**Tech Stack:** Node.js 20 ESM, Express 5, Prisma 7, PostgreSQL + pgvector, `ws` WebSocket, Google Gemini (with fallback)

---

## Global Constraints

- ESM only — `import`/`export`, no `require()`
- No demo-specific code anywhere — no `DEMO_COMPANY_BUNDLE`, no `payload === 'demo'`, no hardcoded org/user/project data
- No switch statements on dataset type in engine code — all dispatch goes through registry
- Tenant isolation mandatory — every write scoped by `workspaceId`; return 400 if header absent
- No inline SQL string concatenation — parameterized queries only (`$1`, `$2`, …)
- All upserts — running the same import twice must never duplicate entities
- Typed errors only — throw `ValidationError`, `AppError`, `NotFoundError` from `src/core/errors/index.js`; never bare `new Error()`
- All Gemini/LLM calls must have a graceful no-key fallback
- Do not touch Prisma schema without migrating — run `npx prisma migrate dev` after schema changes
- No comments explaining what code does — only WHY (non-obvious constraints, fallbacks, security invariants)
- No unused variables — remove rather than prefix with `_`
- Progress events via existing `broadcastToWorkspace(workspaceId, eventType, data)` from `src/services/socketService.js`
- Import `{ prisma }` (named) from `src/core/config/prisma.js` — default import returns undefined
- Engine version string: `'2.0'`; schema version floor: `'1.0'`

```js
// shared helper — avoids repeating validation logic per type
function makeValidator(requiredFields) {
  return function validate(records) {
    const errors = [];
    const warnings = [];
    const ids = new Set();
    for (const [i, rec] of (records ?? []).entries()) {
      for (const f of requiredFields) {
        if (rec[f] == null || rec[f] === '') errors.push(`[${i}] missing required field "${f}"`);
      }
      const id = rec.id ?? rec.slug ?? null;
      if (id) {
        if (ids.has(String(id))) warnings.push(`Duplicate id "${id}" at index ${i}`);
        ids.add(String(id));
      }
    }
    return { valid: errors.length === 0, errors, warnings };
  };
}

// shared helper — no-op normalizer (identity)
function identity(records) { return records ?? []; }

// shared helper — returns empty nodes/edges (for types that don't build graph entries)
function noGraph(records, globalIdMap) { return { nodes: [], edges: [] }; }

// shared helper — returns empty vector items
function noVector(records) { return []; }
```

- [ ] **Step 1: Create manifestParser.js**

```js
// src/core/workspaceLifecycle/manifestParser.js
import { hasDatasetType } from '../workspaceLifecycle/datasetRegistry.js';

export const ENGINE_VERSION = '2.0';
export const SCHEMA_VERSION_FLOOR = '1.0';

// Returns true when schemaVersion satisfies minimum floor (semver major.minor comparison)
function isVersionCompatible(version) {
  if (!version) return false;
  const [major, minor = 0] = version.split('.').map(Number);
  const [floorMaj, floorMin = 0] = SCHEMA_VERSION_FLOOR.split('.').map(Number);
  return major > floorMaj || (major === floorMaj && minor >= floorMin);
}

export function getCompatibilityStatus(schemaVersion) {
  if (!schemaVersion) return 'INCOMPATIBLE';
  if (!isVersionCompatible(schemaVersion)) return 'INCOMPATIBLE';
  // Mark 1.x as DEPRECATED once 2.x schema floor is established
  return 'COMPATIBLE';
}

export function parseManifest(raw) {
  const errors = [];

  if (!raw || typeof raw !== 'object') {
    return { valid: false, errors: ['Manifest must be a JSON object'], schemaVersion: null, engineVersion: null, organization: null, datasets: [], rawManifest: raw };
  }

  const { schemaVersion, engineVersion = null, organization, datasets = [] } = raw;

  if (!schemaVersion) errors.push('manifest.schemaVersion is required');
  else if (getCompatibilityStatus(schemaVersion) === 'INCOMPATIBLE') {
    errors.push(`schemaVersion "${schemaVersion}" is below minimum supported floor "${SCHEMA_VERSION_FLOOR}"`);
  }

  if (!organization?.name) errors.push('manifest.organization.name is required');
  if (!organization?.slug) errors.push('manifest.organization.slug is required');

  if (!Array.isArray(datasets) || datasets.length === 0) {
    errors.push('manifest.datasets must be a non-empty array');
  }

  const parsedDatasets = [];
  const warnings = [];
  for (const [i, entry] of (datasets ?? []).entries()) {
    if (!entry.type) { errors.push(`datasets[${i}] missing required field "type"`); continue; }
    if (!hasDatasetType(entry.type)) {
      warnings.push(`datasets[${i}] type "${entry.type}" is unknown — will be skipped`);
    }
    parsedDatasets.push({ type: entry.type, file: entry.file ?? null });
  }

  return {
    valid: errors.length === 0,
    errors,
    warnings,
    schemaVersion: schemaVersion ?? null,
    engineVersion: engineVersion ?? null,
    organization: organization ?? null,
    datasets: parsedDatasets,
    rawManifest: raw,
  };
}
```

**Implementation rules:**

Each stage:
1. Calls `ctx.emit(stageName, 'started', message)` at top
2. Does its work
3. Calls `ctx.emit(stageName, 'completed', message, data)` at bottom
4. On error: calls `ctx.emit(stageName, 'failed', err.message)` then re-throws

WS event types to emit: `LIFECYCLE_STAGE_STARTED`, `LIFECYCLE_STAGE_COMPLETED`, `LIFECYCLE_STAGE_FAILED`
(`ctx.emit` wraps `broadcastToWorkspace(ctx.workspaceId, eventType, { stage, status, message, data, importId: ctx.importId })`)

**runLifecycleOperation implementation:**

```js
import { nanoid } from 'nanoid';  // available in Node 20 via crypto.randomUUID() — use crypto instead
import { parseManifest, ENGINE_VERSION } from './manifestParser.js';
import { ValidationError } from '../../core/errors/index.js';
import { prisma } from '../../core/config/prisma.js';
import { broadcastToWorkspace } from '../../services/socketService.js';

export async function runLifecycleOperation(operation, workspaceId, { manifest: rawManifest, datasets = {} } = {}) {
  if (!OPERATION_STAGES[operation]) throw new ValidationError(`Unknown operation "${operation}"`);

  const importId = `IMP-${crypto.randomUUID().slice(0,8).toUpperCase()}`;
  const parsedManifest = (operation === 'REFRESH')
    ? { valid: true, errors: [], warnings: [], schemaVersion: '1.0', organization: { name: 'workspace', slug: workspaceId }, datasets: [], rawManifest: {} }
    : parseManifest(rawManifest);

  if (!parsedManifest.valid) throw new ValidationError(`Invalid manifest: ${parsedManifest.errors.join('; ')}`);

  // Create durable ImportRecord at start
  const record = await prisma.importRecord.create({
    data: {
      importId,
      workspaceId,
      operation,
      status: 'RUNNING',
      schemaVersion: parsedManifest.schemaVersion,
      engineVersion: ENGINE_VERSION,
      startedAt: new Date(),
    },
  });

  broadcastToWorkspace(workspaceId, 'LIFECYCLE_STARTED', { importId, operation });

  const ctx = {
    workspaceId,
    orgId: null,
    manifest: parsedManifest,
    datasets,
    normalized: {},
    globalIdMap: new Map(),
    resolvedEdges: [],
    workspace: null,
    graphMetrics: { nodes: 0, edges: 0 },
    memoryCount: 0,
    vectorCount: 0,
    operation,
    importId,
    emit: (stage, status, message, data) => {
      const evType = status === 'started' ? 'LIFECYCLE_STAGE_STARTED'
                   : status === 'completed' ? 'LIFECYCLE_STAGE_COMPLETED'
                   : 'LIFECYCLE_STAGE_FAILED';
      broadcastToWorkspace(workspaceId, evType, { stage, status, message, importId, ...data });
    },
  };

  const stageErrors = [];
  for (const stageFn of OPERATION_STAGES[operation]) {
    try {
      await stageFn(ctx);
    } catch (err) {
      stageErrors.push({ stage: stageFn.name, error: err.message });
      // stageValidate failure is fatal; others are best-effort for post-import stages
      if (stageFn.name === 'stageValidate' || stageFn.name === 'stageBootstrapWorkspace') break;
    }
  }

  const status = stageErrors.length === 0 ? 'COMPLETED' : 'FAILED';
  const statistics = {
    datasets: Object.keys(datasets).map(t => ({ type: t, count: (datasets[t] ?? []).length })),
  };

  const updated = await prisma.importRecord.update({
    where: { id: record.id },
    data: {
      status,
      orgId: ctx.orgId ?? null,
      statistics,
      errors: stageErrors.length > 0 ? stageErrors : null,
      warnings: parsedManifest.warnings?.length ? parsedManifest.warnings : null,
      graphMetrics: ctx.graphMetrics,
      completedAt: new Date(),
    },
  });

  broadcastToWorkspace(workspaceId, status === 'COMPLETED' ? 'LIFECYCLE_COMPLETED' : 'LIFECYCLE_FAILED', {
    importId, operation, status, graphMetrics: ctx.graphMetrics, statistics, errors: stageErrors,
  });

  return updated;
}
```

**validateOnly implementation:**

```js
export async function validateOnly(workspaceId, { manifest: rawManifest, datasets = {} } = {}) {
  const parsed = parseManifest(rawManifest);
  if (!parsed.valid) return { valid: false, manifestErrors: parsed.errors, perType: {}, brokenReferences: [], warnings: parsed.warnings ?? [] };

  const perType = {};
  const allIds = new Set();
  const brokenRefs = [];

  for (const entry of parsed.datasets) {
    const { type } = entry;
    const handler = getDatasetHandler(type);
    if (!handler) { perType[type] = { valid: true, warnings: [`Unknown type "${type}" — skipped`], errors: [] }; continue; }
    const records = datasets[type] ?? [];
    const result = handler.validator(records);
    perType[type] = result;
    // collect all IDs for cross-dataset reference check
    for (const rec of records) { if (rec.id) allIds.add(String(rec.id)); }
  }

  // Cross-dataset broken reference check (relationships dataset)
  for (const rel of datasets.relationships ?? []) {
    if (rel.sourceId && !allIds.has(String(rel.sourceId))) brokenRefs.push(`Broken ref: sourceId "${rel.sourceId}"`);
    if (rel.targetId && !allIds.has(String(rel.targetId))) brokenRefs.push(`Broken ref: targetId "${rel.targetId}"`);
  }

  const allValid = Object.values(perType).every(r => r.valid) && brokenRefs.length === 0;
  return { valid: allValid, manifestErrors: [], perType, brokenReferences: brokenRefs, warnings: parsed.warnings ?? [] };
}
```
